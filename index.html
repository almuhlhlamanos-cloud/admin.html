-- ========================================== -- NextTrade Database Setup -- ========================================== CREATE EXTENSION IF NOT EXISTS pgcrypto; -- 1. User profiles CREATE TABLE IF NOT EXISTS public.nt_profiles ( user_id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE, full_name TEXT NOT NULL DEFAULT '', email TEXT, verification_status TEXT NOT NULL DEFAULT 'unverified' CHECK (verification_status IN ('unverified', 'pending', 'verified', 'rejected')), created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); -- 2. Wallets CREATE TABLE IF NOT EXISTS public.nt_wallets ( user_id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE, balance NUMERIC(18,2) NOT NULL DEFAULT 0 CHECK (balance >= 0), updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); -- 3. Admin accounts CREATE TABLE IF NOT EXISTS public.nt_admins ( user_id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); -- 4. Deposit and withdrawal requests CREATE TABLE IF NOT EXISTS public.nt_transactions ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE, transaction_type TEXT NOT NULL CHECK (transaction_type IN ('deposit', 'withdrawal')), amount NUMERIC(18,2) NOT NULL CHECK (amount > 0), payment_reference TEXT, status TEXT NOT NULL DEFAULT 'pending' CHECK (status IN ('pending', 'approved', 'rejected')), admin_note TEXT, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), reviewed_at TIMESTAMPTZ ); -- 5. Investments CREATE TABLE IF NOT EXISTS public.nt_investments ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE, plan_amount NUMERIC(18,2) NOT NULL CHECK (plan_amount > 0), status TEXT NOT NULL DEFAULT 'pending' CHECK (status IN ('pending', 'active', 'completed', 'rejected', 'cancelled')), created_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); -- 6. Identity verification requests CREATE TABLE IF NOT EXISTS public.nt_kyc_requests ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE, full_name TEXT NOT NULL, front_image_path TEXT NOT NULL, back_image_path TEXT NOT NULL, status TEXT NOT NULL DEFAULT 'pending' CHECK (status IN ('pending', 'approved', 'rejected')), admin_note TEXT, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), reviewed_at TIMESTAMPTZ ); -- ========================================== -- Admin verification function -- ========================================== CREATE OR REPLACE FUNCTION public.nt_is_admin() RETURNS BOOLEAN LANGUAGE SQL STABLE SECURITY DEFINER SET search_path = '' AS $$ SELECT EXISTS ( SELECT 1 FROM public.nt_admins WHERE user_id = (SELECT auth.uid()) ); $$; -- ========================================== -- Automatically create profile and wallet -- when a user signs up -- ========================================== CREATE OR REPLACE FUNCTION public.nt_handle_new_user() RETURNS TRIGGER LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $$ BEGIN INSERT INTO public.nt_profiles (user_id, full_name, email) VALUES ( NEW.id, COALESCE(NEW.raw_user_meta_data ->> 'full_name', ''), NEW.email ) ON CONFLICT (user_id) DO NOTHING; INSERT INTO public.nt_wallets (user_id) VALUES (NEW.id) ON CONFLICT (user_id) DO NOTHING; RETURN NEW; END; $$; DROP TRIGGER IF EXISTS nt_on_auth_user_created ON auth.users; CREATE TRIGGER nt_on_auth_user_created AFTER INSERT ON auth.users FOR EACH ROW EXECUTE FUNCTION public.nt_handle_new_user(); -- ========================================== -- Enable Row Level Security -- ========================================== ALTER TABLE public.nt_profiles ENABLE ROW LEVEL SECURITY; ALTER TABLE public.nt_wallets ENABLE ROW LEVEL SECURITY; ALTER TABLE public.nt_admins ENABLE ROW LEVEL SECURITY; ALTER TABLE public.nt_transactions ENABLE ROW LEVEL SECURITY; ALTER TABLE public.nt_investments ENABLE ROW LEVEL SECURITY; ALTER TABLE public.nt_kyc_requests ENABLE ROW LEVEL SECURITY; -- ========================================== -- Profiles policies -- ========================================== DROP POLICY IF EXISTS nt_profiles_read ON public.nt_profiles; CREATE POLICY nt_profiles_read ON public.nt_profiles FOR SELECT TO authenticated USING (user_id = (SELECT auth.uid()) OR public.nt_is_admin()); DROP POLICY IF EXISTS nt_profiles_insert ON public.nt_profiles; CREATE POLICY nt_profiles_insert ON public.nt_profiles FOR INSERT TO authenticated WITH CHECK (user_id = (SELECT auth.uid())); DROP POLICY IF EXISTS nt_profiles_update ON public.nt_profiles; CREATE POLICY nt_profiles_update ON public.nt_profiles FOR UPDATE TO authenticated USING (user_id = (SELECT auth.uid()) OR public.nt_is_admin()) WITH CHECK (user_id = (SELECT auth.uid()) OR public.nt_is_admin()); -- ========================================== -- Wallet policies -- Users can read their own balance. -- Balance changes must be performed by a -- trusted server-side transaction process. -- ========================================== DROP POLICY IF EXISTS nt_wallets_read ON public.nt_wallets; CREATE POLICY nt_wallets_read ON public.nt_wallets FOR SELECT TO authenticated USING (user_id = (SELECT auth.uid()) OR public.nt_is_admin()); -- ========================================== -- Transaction policies -- ========================================== DROP POLICY IF EXISTS nt_transactions_read ON public.nt_transactions; CREATE POLICY nt_transactions_read ON public.nt_transactions FOR SELECT TO authenticated USING (user_id = (SELECT auth.uid()) OR public.nt_is_admin()); DROP POLICY IF EXISTS nt_transactions_insert ON public.nt_transactions; CREATE POLICY nt_transactions_insert ON public.nt_transactions FOR INSERT TO authenticated WITH CHECK ( user_id = (SELECT auth.uid()) AND status = 'pending' AND amount > 0 ); DROP POLICY IF EXISTS nt_transactions_admin_update ON public.nt_transactions; CREATE POLICY nt_transactions_admin_update ON public.nt_transactions FOR UPDATE TO authenticated USING (public.nt_is_admin()) WITH CHECK (public.nt_is_admin()); -- ========================================== -- Investment policies -- ========================================== DROP POLICY IF EXISTS nt_investments_read ON public.nt_investments; CREATE POLICY nt_investments_read ON public.nt_investments FOR SELECT TO authenticated USING (user_id = (SELECT auth.uid()) OR public.nt_is_admin()); DROP POLICY IF EXISTS nt_investments_insert ON public.nt_investments; CREATE POLICY nt_investments_insert ON public.nt_investments FOR INSERT TO authenticated WITH CHECK ( user_id = (SELECT auth.uid()) AND status = 'pending' AND plan_amount > 0 ); DROP POLICY IF EXISTS nt_investments_admin_update ON public.nt_investments; CREATE POLICY nt_investments_admin_update ON public.nt_investments FOR UPDATE TO authenticated USING (public.nt_is_admin()) WITH CHECK (public.nt_is_admin()); -- ========================================== -- KYC policies -- ========================================== DROP POLICY IF EXISTS nt_kyc_read ON public.nt_kyc_requests; CREATE POLICY nt_kyc_read ON public.nt_kyc_requests FOR SELECT TO authenticated USING (user_id = (SELECT auth.uid()) OR public.nt_is_admin()); DROP POLICY IF EXISTS nt_kyc_insert ON public.nt_kyc_requests; CREATE POLICY nt_kyc_insert ON public.nt_kyc_requests FOR INSERT TO authenticated WITH CHECK ( user_id = (SELECT auth.uid()) AND status = 'pending' ); DROP POLICY IF EXISTS nt_kyc_admin_update ON public.nt_kyc_requests; CREATE POLICY nt_kyc_admin_update ON public.nt_kyc_requests FOR UPDATE TO authenticated USING (public.nt_is_admin()) WITH CHECK (public.nt_is_admin()); -- ========================================== -- Indexes for faster queries -- ========================================== CREATE INDEX IF NOT EXISTS nt_transactions_user_idx ON public.nt_transactions (user_id, created_at DESC); CREATE INDEX IF NOT EXISTS nt_transactions_status_idx ON public.nt_transactions (status); CREATE INDEX IF NOT EXISTS nt_investments_user_idx ON public.nt_investments (user_id, created_at DESC); CREATE INDEX IF NOT EXISTS nt_kyc_user_idx ON public.nt_kyc_requests (user_id, created_at DESC); -- ========================================== -- Done -- ==========================================
